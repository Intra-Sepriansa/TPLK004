2.1 SISTEM POIN KEHADIRAN
--------------------------

POIN PER STATUS KEHADIRAN:
┌─────────────────────────────────────────────────────────────────────────┐
│  Status          │  Poin  │  Keterangan                                 │
├─────────────────────────────────────────────────────────────────────────┤
│  present         │  100   │  Hadir tepat waktu                          │
│  late            │  75    │  Hadir terlambat (< 15 menit)               │
│  permit          │  50    │  Izin dengan surat resmi                    │
│  sick            │  50    │  Sakit dengan surat dokter                  │
│  absent          │  0     │  Tidak hadir tanpa keterangan               │
│  rejected        │  0     │  Presensi ditolak (fraud/invalid)           │
└─────────────────────────────────────────────────────────────────────────┘

2.2 FORMULA PERHITUNGAN NILAI AKHIR
------------------------------------

STEP 1: Hitung Total Poin
Total Poin = Σ (Poin per Pertemuan)

STEP 2: Hitung Maksimum Poin Possible
Max Poin = Total Pertemuan × 100

STEP 3: Hitung Persentase Kehadiran
Attendance Rate = (Total Poin / Max Poin) × 100

STEP 4: Hitung Rata-rata Poin
Average Points = Total Poin / Total Pertemuan

STEP 5: Konversi ke Nilai Huruf (Grade Letter)
┌─────────────────────────────────────────────────────────────────────────┐
│  Attendance Rate  │  Grade  │  Keterangan                               │
├─────────────────────────────────────────────────────────────────────────┤
│  >= 90%           │  A      │  Excellent - Sangat Baik                  │
│  80% - 89%        │  B      │  Good - Baik                              │
│  70% - 79%        │  C      │  Satisfactory - Cukup                     │
│  60% - 69%        │  D      │  Poor - Kurang                            │
│  < 60%            │  E      │  Fail - Gagal                             │
└─────────────────────────────────────────────────────────────────────────┘

STEP 6: Tentukan Kelayakan UAS
can_take_uas = (attendance_rate >= 75) OR (valid_permits >= required_permits)

2.3 SPECIAL CASES
-----------------
- Mahasiswa Join Late: Hanya hitung pertemuan setelah join date
- Pertemuan Dibatalkan: Tidak dihitung dalam Total Pertemuan
- Override Manual: Simpan audit trail lengkap
- Mahasiswa Cuti/Drop: Status inactive, nilai di-freeze
- Remedial/Makeup: Status berubah dari absent → permit

// Efficient query for grades calculation
$grades = DB::table('attendance_logs')
    ->join('mahasiswa', 'attendance_logs.mahasiswa_id', '=', 'mahasiswa.id')
    ->join('sessions', 'attendance_logs.session_id', '=', 'sessions.id')
    ->where('sessions.course_id', $courseId)
    ->select(
        'mahasiswa.id as mahasiswa_id',
        'mahasiswa.nama',
        'mahasiswa.nim',
        DB::raw('COUNT(DISTINCT sessions.id) as total_sessions'),
        DB::raw('SUM(CASE WHEN attendance_logs.status = "present" THEN 1 ELSE 0 END) as attended_sessions'),
        DB::raw('SUM(CASE 
            WHEN attendance_logs.status = "present" THEN 100
            WHEN attendance_logs.status = "late" THEN 75
            WHEN attendance_logs.status = "permit" THEN 50
            WHEN attendance_logs.status = "sick" THEN 50
            ELSE 0
        END) as total_points'),
        DB::raw('ROUND((SUM(CASE 
            WHEN attendance_logs.status = "present" THEN 100
            WHEN attendance_logs.status = "late" THEN 75
            WHEN attendance_logs.status = "permit" THEN 50
            WHEN attendance_logs.status = "sick" THEN 50
            ELSE 0
        END) / (COUNT(DISTINCT sessions.id) * 100)) * 100, 2) as attendance_rate')
    )
    ->groupBy('mahasiswa.id', 'mahasiswa.nama', 'mahasiswa.nim')
    ->get();

// Add grade letter and UAS eligibility
$grades = $grades->map(function($grade) {
    $grade->average_points = round($grade->total_points / $grade->total_sessions, 2);
    
    // Determine grade letter
    if ($grade->attendance_rate >= 90) {
        $grade->grade_letter = 'A';
    } elseif ($grade->attendance_rate >= 80) {
        $grade->grade_letter = 'B';
    } elseif ($grade->attendance_rate >= 70) {
        $grade->grade_letter = 'C';
    } elseif ($grade->attendance_rate >= 60) {
        $grade->grade_letter = 'D';
    } else {
        $grade->grade_letter = 'E';
    }
    
    // Determine UAS eligibility
    $grade->can_take_uas = $grade->attendance_rate >= 75;
    
    return $grade;
});

public function override(Request $request)
{
    $validated = $request->validate([
        'log_id' => 'required|exists:attendance_logs,id',
        'status' => 'required|in:present,late,permit,sick,absent,rejected',
        'reason' => 'required|string|min:10'
    ]);
    
    $log = AttendanceLog::find($validated['log_id']);
    $oldStatus = $log->status;
    
    // Update status
    $log->status = $validated['status'];
    $log->save();
    
    // Save audit trail
    DB::table('grade_audit_trail')->insert([
        'log_id' => $log->id,
        'mahasiswa_id' => $log->mahasiswa_id,
        'old_status' => $oldStatus,
        'new_status' => $validated['status'],
        'reason' => $validated['reason'],
        'changed_by' => auth()->id(),
        'changed_at' => now()
    ]);
    
    // Recalculate grades
    $this->recalculateGrades($log->mahasiswa_id);
    
    // Send notification
    $this->sendNotification($log->mahasiswa_id, 
        "Status kehadiran Anda telah diupdate dari {$oldStatus} menjadi {$validated['status']}. Alasan: {$validated['reason']}"
    );
    
    return back()->with('success', 'Status berhasil diupdate');
}

const calculateStatistics = (grades: Grade[]) => {
    const stats = {
        total_students: grades.length,
        total_sessions: grades[0]?.total_sessions || 0,
        average_attendance_rate: 0,
        students_at_risk: 0,
        grade_distribution: { A: 0, B: 0, C: 0, D: 0, E: 0 },
        attendance_ranges: {
            excellent: 0,  // >= 90%
            good: 0,       // 80-89%
            warning: 0,    // 60-79%
            critical: 0    // < 60%
        }
    };
    
    let totalRate = 0;
    
    grades.forEach(grade => {
        // Sum attendance rate
        totalRate += grade.attendance_rate;
        
        // Count at-risk students
        if (!grade.can_take_uas) {
            stats.students_at_risk++;
        }
        
        // Count grade distribution
        stats.grade_distribution[grade.grade_letter]++;
        
        // Count attendance ranges
        if (grade.attendance_rate >= 90) {
            stats.attendance_ranges.excellent++;
        } else if (grade.attendance_rate >= 80) {
            stats.attendance_ranges.good++;
        } else if (grade.attendance_rate >= 60) {
            stats.attendance_ranges.warning++;
        } else {
            stats.attendance_ranges.critical++;
        }
    });
    
    // Calculate average
    stats.average_attendance_rate = grades.length > 0 
        ? Math.round(totalRate / grades.length) 
        : 0;
    
    return stats;
};

7.6 FILTER GRADES HELPER
-------------------------

const filterGrades = (
    grades: Grade[],
    searchQuery: string,
    filterGrade: string,
    filterUAS: string,
    filterSession: string,
    sortBy: string,
    sortOrder: 'asc' | 'desc'
) => {
    let filtered = grades.filter(g => {
        // Search filter
        const matchSearch = g.nama.toLowerCase().includes(searchQuery.toLowerCase()) || 
                           g.nim.includes(searchQuery);
        
        // Grade filter
        const matchGrade = filterGrade === 'all' || g.grade_letter === filterGrade;
        
        // UAS filter
        const matchUAS = filterUAS === 'all' || 
                       (filterUAS === 'can' && g.can_take_uas) || 
                       (filterUAS === 'cannot' && !g.can_take_uas);
        
        // Session filter (check if student attended specific session)
        let matchSession = true;
        if (filterSession !== 'all') {
            const sessionNum = parseInt(filterSession);
            const sessionDetail = g.details.find(d => d.meeting === sessionNum);
            matchSession = sessionDetail !== undefined;
        }
        
        return matchSearch && matchGrade && matchUAS && matchSession;
    });
    
    // Sort
    filtered.sort((a, b) => {
        let comparison = 0;
        switch (sortBy) {
            case 'name':
                comparison = a.nama.localeCompare(b.nama);
                break;
            case 'nim':
                comparison = a.nim.localeCompare(b.nim);
                break;
            case 'rate':
                comparison = a.attendance_rate - b.attendance_rate;
                break;
            case 'grade':
                const gradeOrder = { 'A': 5, 'B': 4, 'C': 3, 'D': 2, 'E': 1 };
                comparison = (gradeOrder[a.grade_letter] || 0) - 
                            (gradeOrder[b.grade_letter] || 0);
                break;
        }
        return sortOrder === 'asc' ? comparison : -comparison;
    });
    
    return filtered;
};

const generateAIInsights = (student: Grade) => {

        });
    } else if (student.attendance_rate >= 60) {
        insights.push({
            type: 'warning',
            title: 'Warning - Needs Improvement',
            message: 'Mahasiswa perlu meningkatkan kehadiran. Lakukan konseling untuk identifikasi kendala.'
        });
    } else {
        insights.push({
            type: 'danger',
            title: 'Critical - Immediate Action Required',
            message: 'Mahasiswa berisiko tidak bisa UAS. Segera hubungi untuk intervensi dan buat action plan.'
        });
    }
    
    // Recommendations
    if (!student.can_take_uas) {
        insights.push({
            type: 'action',

            actions: [
                'Hubungi mahasiswa untuk konseling individual',
                'Identifikasi kendala yang menyebabkan rendahnya kehadiran',
                'Buat agreement tertulis untuk perbaikan kehadiran',
                'Monitor progress setiap pertemuan',
                'Pertimbangkan remedial atau makeup assignments'
            ]
        });
    }
    
    // Pattern detection
    const recentAttendance = student.details.slice(-5);
    const recentAbsences = recentAttendance.filter(d => d.status === 'absent').length;
    if (recentAbsences >= 3) {
        insights.push({

            message: `Mahasiswa tidak hadir ${recentAbsences} kali dalam 5 pertemuan terakhir. Kemungkinan ada masalah serius.`
        });
    }
    
    return insights;
};
